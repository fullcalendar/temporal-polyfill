# Node <=22 does not provide Intl.DurationFormat. Duration#toLocaleString still
# has an ISO-string fallback for useful behavior, but this ECMA-402 test compares
# its output against new Intl.DurationFormat(...).format(...), which is only
# available starting in newer Node releases.
intl402/Temporal/Duration/prototype/toLocaleString/returns-same-results-as-DurationFormat.js

# ENV-ISSUE
# Node <=22 host Intl.DateTimeFormat has a Hebrew/lunisolar formatToParts
# mismatch with Temporal's calendar fields. Node 24+ no longer reproduces this
# failure, so keep this out of the general shim expected-failure list.
intl402/DateTimeFormat/prototype/formatToParts/compare-to-temporal-lunisolar.js

# We intentionally avoid one-off IANA-link workarounds for host Intl data.
